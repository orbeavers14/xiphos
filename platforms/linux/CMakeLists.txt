cmake_minimum_required(VERSION 3.16)
project(xiphos_linux_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
option(XIPHOS_BUILD_GTK4_PREVIEW "Build the GTK4/WebKitGTK desktop preview" ON)
if(XIPHOS_BUILD_GTK4_PREVIEW)
  find_package(PNG 1.6 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.6)
  pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkitgtk-6.0>=2.40)
endif()
set(XIPHOS_BUILD_NATIVE_CORE ON)
add_subdirectory(../../src/core core)
add_library(linux_reader_model reader_model.cpp study_model.cpp reader_document.cpp)
target_link_libraries(linux_reader_model PUBLIC Xiphos::Core)
target_include_directories(linux_reader_model PUBLIC .)
if(XIPHOS_BUILD_GTK4_PREVIEW)
  add_executable(xiphos-linux-preview reader_app.cpp study_ui.cpp)
  target_link_libraries(xiphos-linux-preview PRIVATE linux_reader_model PkgConfig::GTK4 PkgConfig::WEBKIT PNG::PNG)
  if(BUILD_TESTING)
    # Opt-in desktop smoke; never part of unattended CTest/hosted runs.
    add_executable(linux_study_ui_smoke tests/study_ui_smoke.cpp study_ui.cpp)
    target_link_libraries(linux_study_ui_smoke PRIVATE linux_reader_model PkgConfig::GTK4 PkgConfig::WEBKIT PNG::PNG)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(linux_study_model_test tests/study_model_test.cpp)
  target_link_libraries(linux_study_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_study_model COMMAND linux_study_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/book-document-presentation-fixture" "${CMAKE_CURRENT_BINARY_DIR}/core/attribute-search-fixture" "${CMAKE_CURRENT_BINARY_DIR}/linux-study-fixture" "${CMAKE_CURRENT_BINARY_DIR}/core/verse-image-source-fixture")
  set_tests_properties(linux_study_model PROPERTIES FIXTURES_REQUIRED "book_document_presentation_fixture;attribute_search_fixture;verse_image_fixture" TIMEOUT 90)
  add_executable(linux_heading_note_model_test tests/heading_note_model_test.cpp)
  target_link_libraries(linux_heading_note_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_heading_note_model COMMAND linux_heading_note_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/heading-snapshots-fixture")
  set_tests_properties(linux_heading_note_model PROPERTIES FIXTURES_REQUIRED heading_snapshots_fixture TIMEOUT 30)
  add_executable(linux_note_model_test tests/note_model_test.cpp)
  target_link_libraries(linux_note_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_note_model COMMAND linux_note_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/note-snapshots-fixture")
  set_tests_properties(linux_note_model PROPERTIES FIXTURES_REQUIRED note_snapshots_fixture TIMEOUT 30)
  add_executable(linux_lexical_model_test tests/lexical_model_test.cpp)
  target_link_libraries(linux_lexical_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_lexical_model COMMAND linux_lexical_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/lexical-snapshots-fixture")
  set_tests_properties(linux_lexical_model PROPERTIES FIXTURES_REQUIRED lexical_snapshots_fixture TIMEOUT 30)
  add_executable(linux_window_model_test tests/window_model_test.cpp)
  target_link_libraries(linux_window_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_window_model COMMAND linux_window_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/module-preferences-fixture")
  set_tests_properties(linux_window_model PROPERTIES FIXTURES_REQUIRED module_preferences_fixture TIMEOUT 30)
  add_executable(linux_preview_model_test tests/preview_model_test.cpp)
  target_link_libraries(linux_preview_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_preview_model COMMAND linux_preview_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/module-preferences-fixture" "${CMAKE_CURRENT_BINARY_DIR}/core/dictionary-fixture" "${CMAKE_CURRENT_BINARY_DIR}/linux-lexicon-profile-fixture")
  set_tests_properties(linux_preview_model PROPERTIES FIXTURES_REQUIRED "module_preferences_fixture;dictionary_fixture" TIMEOUT 30)
  add_executable(linux_options_model_test tests/options_model_test.cpp)
  target_link_libraries(linux_options_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_options_model COMMAND linux_options_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/module-preferences-fixture" "${CMAKE_CURRENT_BINARY_DIR}/linux-profile-fixture")
  set_tests_properties(linux_options_model PROPERTIES FIXTURES_REQUIRED module_preferences_fixture TIMEOUT 30)
  add_executable(linux_reader_model_test tests/reader_model_test.cpp)
  target_link_libraries(linux_reader_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_reader_model COMMAND linux_reader_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/study-passage-fixture" "${CMAKE_CURRENT_BINARY_DIR}/empty-library")
  set_tests_properties(linux_reader_model PROPERTIES FIXTURES_REQUIRED study_passage_fixture TIMEOUT 30)
endif()
