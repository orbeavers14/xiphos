cmake_minimum_required(VERSION 3.16)
project(xiphos_linux_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.6)
pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkitgtk-6.0>=2.40)
set(XIPHOS_BUILD_NATIVE_CORE ON)
add_subdirectory(../../src/core core)
add_library(linux_reader_model reader_model.cpp)
target_link_libraries(linux_reader_model PUBLIC Xiphos::Core)
target_include_directories(linux_reader_model PUBLIC .)
add_executable(xiphos-linux-preview reader_app.cpp)
target_link_libraries(xiphos-linux-preview PRIVATE linux_reader_model PkgConfig::GTK4 PkgConfig::WEBKIT)
if(BUILD_TESTING)
  add_executable(linux_preview_model_test tests/preview_model_test.cpp)
  target_link_libraries(linux_preview_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_preview_model COMMAND linux_preview_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/module-preferences-fixture" "${CMAKE_CURRENT_BINARY_DIR}/core/dictionary-fixture" "${CMAKE_CURRENT_BINARY_DIR}/linux-lexicon-profile-fixture")
  set_tests_properties(linux_preview_model PROPERTIES FIXTURES_REQUIRED "module_preferences_fixture;dictionary_fixture" TIMEOUT 30)
  add_executable(linux_options_model_test tests/options_model_test.cpp)
  target_link_libraries(linux_options_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_options_model COMMAND linux_options_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/module-preferences-fixture" "${CMAKE_CURRENT_BINARY_DIR}/linux-profile-fixture")
  set_tests_properties(linux_options_model PROPERTIES FIXTURES_REQUIRED module_preferences_fixture TIMEOUT 30)
  add_executable(linux_reader_model_test tests/reader_model_test.cpp)
  target_link_libraries(linux_reader_model_test PRIVATE linux_reader_model)
  add_test(NAME linux_reader_model COMMAND linux_reader_model_test "${CMAKE_CURRENT_BINARY_DIR}/core/study-passage-fixture" "${CMAKE_CURRENT_BINARY_DIR}/empty-library")
  set_tests_properties(linux_reader_model PROPERTIES FIXTURES_REQUIRED study_passage_fixture TIMEOUT 30)
endif()
