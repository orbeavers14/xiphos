# Reuse the legacy dependency when built inside Xiphos. Standalone builds
# also accept a prebuilt SWORD SDK on platforms without pkg-config.
if(NOT TARGET PkgConfig::Sword)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(Sword QUIET IMPORTED_TARGET GLOBAL "sword>=1.8.1")
  endif()
endif()
if(TARGET PkgConfig::Sword)
  set(XIPHOS_SWORD_TARGET PkgConfig::Sword)
else()
  find_path(SWORD_INCLUDE_DIR swmgr.h PATH_SUFFIXES sword)
  find_library(SWORD_LIBRARY NAMES sword)
  if(NOT SWORD_INCLUDE_DIR OR NOT SWORD_LIBRARY)
    message(FATAL_ERROR "SWORD >=1.8.1 headers and library are required")
  endif()
  add_library(xiphos_sword UNKNOWN IMPORTED)
  set_target_properties(xiphos_sword PROPERTIES IMPORTED_LOCATION "${SWORD_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SWORD_INCLUDE_DIR}")
  set(XIPHOS_SWORD_TARGET xiphos_sword)
endif()

  set(abbreviation_actual "")
  foreach(handler main_add_abbreviation main_clear_abbreviations main_name_to_abbrev main_abbrev_to_name)
    if(handler MATCHES "^main_(name_to|abbrev_to)")
      set(handler_signature "\nconst char *${handler}(")
    else()
      set(handler_signature "\nvoid ${handler}(")
    endif()
    string(FIND "${abbreviation_source}" "${handler_signature}" handler_start)
    if(handler_start LESS 0)
      message(FATAL_ERROR "Cannot find retained abbreviation helper ${handler}")
    endif()

      message(FATAL_ERROR "Cannot delimit retained abbreviation helper ${handler}")
    endif()
    math(EXPR handler_length "${handler_end} + 3")
    string(SUBSTRING "${handler_tail}" 0 ${handler_length} handler_body)
    string(APPEND abbreviation_actual "${handler_body}\n")
  endforeach()

# SWORD does not expose its regex backend's case bit in the public API.
# GNU/ICU SDK builds use 2; C++11-regex SDK builds use 1. Validate the matching
# SDK build value with core_search before claiming case-sensitive parity.
set(XIPHOS_SWORD_SEARCH_ICASE_FLAG "2" CACHE STRING "Case-insensitive search bit matching the SWORD SDK regex backend (1 or 2)")
if(NOT XIPHOS_SWORD_SEARCH_ICASE_FLAG MATCHES "^[12]$")
  message(FATAL_ERROR "SWORD search case bit must be 1 (C++11 regex) or 2 (GNU/ICU), matching the linked SDK")
endif()
