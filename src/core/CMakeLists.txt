# Reuse the legacy dependency when built inside Xiphos. Standalone builds
# also accept a prebuilt SWORD SDK on platforms without pkg-config.
if(NOT TARGET PkgConfig::Sword)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(Sword QUIET IMPORTED_TARGET GLOBAL "sword>=1.8.1")
  endif()
endif()
if(TARGET PkgConfig::Sword)
  set(XIPHOS_SWORD_TARGET PkgConfig::Sword)
else()
  find_path(SWORD_INCLUDE_DIR swmgr.h PATH_SUFFIXES sword)
  find_library(SWORD_LIBRARY NAMES sword)
  if(NOT SWORD_INCLUDE_DIR OR NOT SWORD_LIBRARY)
    message(FATAL_ERROR "SWORD >=1.8.1 headers and library are required")
  endif()
  add_library(xiphos_sword UNKNOWN IMPORTED)
  set_target_properties(xiphos_sword PROPERTIES IMPORTED_LOCATION "${SWORD_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SWORD_INCLUDE_DIR}")
  set(XIPHOS_SWORD_TARGET xiphos_sword)
endif()
set(XIPHOS_SWORD_EXTRA_LIBRARIES "" CACHE STRING "Extra libraries required by a static SWORD SDK (semicolon-separated)")
# GLib is already a legacy dependency. Reuse its Unicode casing without GTK.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(Glib QUIET IMPORTED_TARGET GLOBAL "glib-2.0>=2.32")
endif()
if(TARGET PkgConfig::Glib)
  set(XIPHOS_GLIB_TARGET PkgConfig::Glib)
else()
  find_path(GLIB_INCLUDE_DIR glib.h PATH_SUFFIXES glib-2.0)
  find_path(GLIB_CONFIG_INCLUDE_DIR glibconfig.h PATH_SUFFIXES glib-2.0/include)
  find_library(GLIB_LIBRARY NAMES glib-2.0)
  if(NOT GLIB_INCLUDE_DIR OR NOT GLIB_CONFIG_INCLUDE_DIR OR NOT GLIB_LIBRARY)
    message(FATAL_ERROR "GLib >=2.32 headers and library are required for legacy Unicode casing")
  endif()
  add_library(xiphos_glib UNKNOWN IMPORTED)
  set_target_properties(xiphos_glib PROPERTIES IMPORTED_LOCATION "${GLIB_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${GLIB_INCLUDE_DIR};${GLIB_CONFIG_INCLUDE_DIR}")
  set(XIPHOS_GLIB_TARGET xiphos_glib)
endif()
set(XIPHOS_GLIB_EXTRA_LIBRARIES "" CACHE STRING "Extra libraries required by a static GLib SDK (semicolon-separated)")
