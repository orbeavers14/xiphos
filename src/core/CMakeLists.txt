if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.16)
  project(xiphos_core LANGUAGES C CXX)
  include(CTest)
  set(XIPHOS_BUILD_NATIVE_CORE ON)
else()
  cmake_minimum_required(VERSION 3.11)
endif()

# Reuse the legacy dependency when built inside Xiphos. Standalone builds
# also accept a prebuilt SWORD SDK on platforms without pkg-config.
if(NOT TARGET PkgConfig::Sword)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(Sword QUIET IMPORTED_TARGET GLOBAL "sword>=1.8.1")
  endif()
endif()
if(TARGET PkgConfig::Sword)
  set(XIPHOS_SWORD_TARGET PkgConfig::Sword)
else()
  find_path(SWORD_INCLUDE_DIR swmgr.h PATH_SUFFIXES sword)
  find_library(SWORD_LIBRARY NAMES sword)
  if(NOT SWORD_INCLUDE_DIR OR NOT SWORD_LIBRARY)
    message(FATAL_ERROR "SWORD >=1.8.1 headers and library are required")
  endif()
  add_library(xiphos_sword UNKNOWN IMPORTED)
  set_target_properties(xiphos_sword PROPERTIES IMPORTED_LOCATION "${SWORD_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${SWORD_INCLUDE_DIR}")
  set(XIPHOS_SWORD_TARGET xiphos_sword)
endif()
set(XIPHOS_SWORD_EXTRA_LIBRARIES "" CACHE STRING "Extra libraries required by a static SWORD SDK (semicolon-separated)")
add_library(xiphos_core_render STATIC render.cpp)
add_library(Xiphos::CoreRender ALIAS xiphos_core_render)
target_include_directories(xiphos_core_render PUBLIC include)
target_compile_features(xiphos_core_render PUBLIC cxx_std_11)
target_link_libraries(xiphos_core_render PUBLIC ${XIPHOS_SWORD_TARGET} ${XIPHOS_SWORD_EXTRA_LIBRARIES})
set_target_properties(xiphos_core_render PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(xiphos_core_history STATIC history.cpp)
add_library(Xiphos::CoreHistory ALIAS xiphos_core_history)
target_include_directories(xiphos_core_history PUBLIC include)
target_compile_features(xiphos_core_history PUBLIC cxx_std_11)
set_target_properties(xiphos_core_history PROPERTIES POSITION_INDEPENDENT_CODE ON)
# Only the small rendering seam is built by the default legacy application.
if(NOT XIPHOS_BUILD_NATIVE_CORE)
  return()
endif()
find_package(Threads REQUIRED)
add_library(xiphos_core STATIC reader.cpp session.cpp content.cpp c_api.cpp)
add_library(Xiphos::Core ALIAS xiphos_core)
target_compile_features(xiphos_core PUBLIC cxx_std_17)
target_include_directories(xiphos_core PUBLIC include)
target_link_libraries(xiphos_core PUBLIC Xiphos::CoreRender Xiphos::CoreHistory Threads::Threads)
set_target_properties(xiphos_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
